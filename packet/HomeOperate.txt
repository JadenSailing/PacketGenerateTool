//家具资源CG和GC包（目的：取服务器MD5值做校验）
//注意：当Res表格里配置的大于128个数据后，1个ResAsk包会回多个ResRet包
//注意：需要在家园类型的地图上，人物正常进入场景后，发送此消息
msg gc GCHomeBieYeStaticResRet
{
	int op_type		//0:非法 1.返回静态资源MD5
	int ResSize		//本消息包里有多少个数据（最大值128）
	int[ResSize] ResType	//资源类型，表格第一列
	char[32][ResSize] ResMD5	//服务器加载Res时计算出的32位MD5串
}

//家具摆放CG
msg cg CGHomePushOperationReq
{
	struct add_item //添加家具结构
	{
		int objTypeAdd
		int isFurniture
		float anchorX
		float anchorY
		float anchorZ
		float yaw
		float scale
		int client_param
		int64 nGuidAdd
	}
	struct del_item //删除家具结构
	{
		int objTypeDel
		int64 nGuidDel
	}
	int	op_type	//0:非法 1:建筑相关 2:清空建筑 3:地形相关 4水面调整 5重置地形(会同时删除建筑列表和水面)
	int64 nHomeGuid = 123456 //家园GUID
	case op_type 1 //既添加又删除
		int		addSize			//添加数量（最大值1024）
		add_item[addSize] add_item //添加的家具
		int		delSize			//删除数量（最大值1024）
		del_item[delSize] del_item //删除的家具
		byte	useMoney = 1 //是否要购买不存在的家居
		byte 	isOfficial = 1 //是否是官方模板
		byte 	index = 255	//模板索引 255表示不使用模板
	endcase
	case op_type 2 //清空地图所有摆放
		//什么也不传
	endcase
	case op_type 3 //地形相关
		int curVersion //当前版本号
		byte isGuanFang //是否是官方模板
		byte index //模板索引 -1表示不使用模板
		int needUpdate = 1 //是否需要透传 默认1
		case index 255
			char[32] md5 //体素数据md5 不需要的时候不传
			int navDataSize //体素数据大小
			bytes[navDataSize] navData //体素数据
		endcase
	endcase
	case op_type 4 //水面调整
		byte useWater //是否启用水面
		float waterLevel //水面高度
	endcase
	case op_type 5 //重置地形
		//什么也不传
	endcase
}

{
	struct add_item //添加家具结构
	{
		int objTypeAdd
		int isFurniture
		float anchorX
		float anchorY
		float anchorZ
		float yaw

		int client_param
		int64 nGuidAdd
	}
	struct del_item //删除家具结构
	{
		int objTypeDel
		int64 nGuidDel
	}

	int op_type		//0:非法 1:摆放成功 2:清除建筑成功 3地表 4水位 对应负值均为失败
	int64 nHomeGuid		//家园GUID
	int64 nOpUserGuid		//操作者GUID
	int		nLogicRet		//游戏逻辑处理的返回值	0:非法 1:成功 -1:资源不存在 -2:anchor坐标非法 
	int		nPushRet		//Push函数的返回值		0:成功 其他:失败
	int		nApplyRet		//Apply函数的返回值		0:成功 其他:失败
	case op_type 1
		int		addSize			//添加数量（最大值1024）
		add_item[addSize] add_item //添加的家具
		int		delSize			//删除数量（最大值1024）
		del_item[delSize] del_item //删除的家具
	endcase
	case op_type 2
	endcase
	case op_type 3
	endcase
	case op_type 4
	endcase
	case op_type 5
	endcase
	case op_type -1 //摆放失败
		int		addSize			//添加数量（最大值1024）
		add_item[addSize] add_item //添加的家具
		int		delSize			//删除数量（最大值1024）
		del_item[delSize] del_item //删除的家具
	endcase
	case op_type -2 //清除建筑失败
	endcase
	case op_type -3 //地表失败
	endcase
	case op_type -4 //水位失败
	endcase
	case op_type -5 //重置地形失败
	endcase
}
